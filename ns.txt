2. Implement Mono alphabetic cipher with the given key and input

import java.util.*;
public class MonoalphabeticCipher {
 public static void main(String[] args) {
 Scanner sc = new Scanner(System.in);
 // Define the standard alphabet
 String alphabet = "abcdefghijklmnopqrstuvwxyz";
 // Example Key: A full 26-character scrambled alphabet
 String key = "zyxwvutsrqponmlkjihgfedcba";
 System.out.print("Enter plaintext: ");
 String plaintext = sc.nextLine().toLowerCase();
 // 1. Create mapping for Encryption
 Map<Character, Character> encryptMap = new HashMap<>();
 for (int i = 0; i < alphabet.length(); i++) {
 encryptMap.put(alphabet.charAt(i), key.charAt(i));
 }
 // 2. Perform Encryption
 StringBuilder ciphertext = new StringBuilder();
 for (char c : plaintext.toCharArray()) {
 ciphertext.append(encryptMap.getOrDefault(c, c));
 }
 System.out.println("Ciphertext: " + ciphertext);
 // 3. Perform Decryption (Reverse Mapping)
 Map<Character, Character> decryptMap = new HashMap<>(); 
for (int i = 0; i < key.length(); i++) {
 decryptMap.put(key.charAt(i), alphabet.charAt(i));
 }
 StringBuilder originalText = new StringBuilder();
 for (char c : ciphertext.toString().toCharArray()) {
 originalText.append(decryptMap.getOrDefault(c, c));
 }
 System.out.println("Decrypted Text: " + originalText);
 sc.close();
 }
}

Ciphertext: yprg yszopr
Decrypted Text: bkit bhalki

3. Implement Poly alphabetic Cipher
public class VigenereCipher {
 public static String encrypt(String text, final String key) {
 StringBuilder result = new StringBuilder();
 text = text.toUpperCase();
 String keyUpper = key.toUpperCase();
 for (int i = 0, j = 0; i < text.length(); i++) {
 char c = text.charAt(i);
 if (c < 'A' || c > 'Z') {
 result.append(c);
 continue;
 }
 result.append((char) ((c + keyUpper.charAt(j) - 2 * 'A') % 26 + 'A'));
 j = ++j % keyUpper.length();
 }
 return result.toString();
 }
 public static String decrypt(String text, final String key) {
 StringBuilder result = new StringBuilder();
 text = text.toUpperCase();
 String keyUpper = key.toUpperCase();
 for (int i = 0, j = 0; i < text.length(); i++) {
 char c = text.charAt(i);
 if (c < 'A' || c > 'Z') {
 result.append(c); 
continue;
 }
 result.append((char) ((c - keyUpper.charAt(j) + 26) % 26 + 'A'));
 j = ++j % keyUpper.length();
 }
 return result.toString();
 }
 public static void main(String[] args) {
 String key = "VIGENERECIPHER";
 String message = "Beware the Jabberwock, my son! The jaws that bite, the claws that
catch!";
 String encryptedMsg = encrypt(message, key);
 System.out.println("Original String: " + message);
 System.out.println("Encrypted message: " + encryptedMsg);
 String decryptedMsg = decrypt(encryptedMsg, key);
 System.out.println("Decrypted message: " + decryptedMsg);
 }
}

import java.util.*;
public class Transposition {
 // Helper to get column order based on key (e.g., "COL" -> {0, 2, 1})
 private static Integer[] getColumnOrder(String key) {
 int n = key.length();
 Integer[] order = new Integer[n];
 for (int i = 0; i < n; i++) order[i] = i;
 Arrays.sort(order, Comparator.comparingInt(i -> key.charAt(i)));
 return order;
 }
 public static String encrypt(String text, String key) {
 int col = key.length();
 int row = (int) Math.ceil((double) text.length() / col);
 char[][] grid = new char[row][col];
 // Fill row-wise
 int k = 0;
 for (int i = 0; i < row; i++) {
 for (int j = 0; j < col; j++) {
 grid[i][j] = (k < text.length()) ? text.charAt(k++) : 'X'; // Padding
 }
 }
 // Read column-wise based on key order
 StringBuilder result = new StringBuilder();
 for (int c : getColumnOrder(key)) { 

for (int r = 0; r < row; r++) result.append(grid[r][c]);
 }
 return result.toString();
 }
 public static void main(String[] args) {
 String msg = "HELLOWORLD";
 String key1 = "BCA";
 String key2 = "CAB";
 // Single Transposition
 String singleEnc = encrypt(msg, key1);
 // Double Transposition (Pass 1 result through Pass 2)
 String doubleEnc = encrypt(singleEnc, key2);
 System.out.println("Original: " + msg);
 System.out.println("Single Transposition: " + singleEnc);
 System.out.println("Double Transposition: " + doubleEnc);
 }
}

10.Implement RSA Algorithm
import javax.crypto.Cipher;
import java.security.KeyPair;
import java.security.KeyPairGenerator;
import java.security.PrivateKey;
import java.security.PublicKey;
import java.util.Base64;
public class RSAExample {
 public static void main(String[] args) throws Exception {
 // 1. Generate RSA Key Pair (2048 bits is standard)
 KeyPairGenerator keyGen = KeyPairGenerator.getInstance("RSA");
 keyGen.initialize(2048);
 KeyPair pair = keyGen.generateKeyPair();
 PublicKey publicKey = pair.getPublic();
 PrivateKey privateKey = pair.getPrivate();
 String originalMessage = "Hello, this is a secret!";
 // 2. Encryption
 Cipher encryptCipher = Cipher.getInstance("RSA/ECB/PKCS1Padding");
 encryptCipher.init(Cipher.ENCRYPT_MODE, publicKey);
 byte[] encryptedBytes = encryptCipher.doFinal(originalMessage.getBytes());
 String encodedMessage = Base64.getEncoder().encodeToString(encryptedBytes);
 // 3. Decryption
 Cipher decryptCipher = Cipher.getInstance("RSA/ECB/PKCS1Padding");
 decryptCipher.init(Cipher.DECRYPT_MODE, privateKey);
byte[] decryptedBytes =
decryptCipher.doFinal(Base64.getDecoder().decode(encodedMessage));
 String decryptedMessage = new String(decryptedBytes);
 System.out.println("Original: " + originalMessage);
 System.out.println("Encrypted (Base64): " + encodedMessage);
 System.out.println("Decrypted: " + decryptedMessage);
 }
}

11.Demonstrate Diffie Hellman Key exchange Algorithm
import java.math.BigInteger;
import java.util.Scanner;
public class DiffieHellmanDemo {
 public static void main(String[] args) {
 Scanner stdin = new Scanner(System.in);
 // 1. Publicly agreed upon parameters (g and p)
 // For demonstration, we use small values. In real applications, these are large primes.
 BigInteger p = new BigInteger("23"); // Prime modulus
 BigInteger g = new BigInteger("5"); // Generator/base
 System.out.println("Publicly agreed Prime (p): " + p);
 System.out.println("Publicly agreed Base (g): " + g);
 // 2. Alice and Bob choose their private keys (secret integers)
 System.out.println("\nEnter Alice's private key (a): ");
 BigInteger a = new BigInteger(stdin.next());
 System.out.println("Enter Bob's private key (b): ");
 BigInteger b = new BigInteger(stdin.next());
 // 3. Alice and Bob calculate and exchange their public keys
 // Alice's public key A = g^a mod p
 BigInteger A = g.modPow(a, p);
 System.out.println("\nAlice's public value (A) sent to Bob: " + A);
 // Bob's public key B = g^b mod p
 BigInteger B = g.modPow(b, p);
 System.out.println("Bob's public value (B) sent to Alice: " + B);
 // 4. Alice and Bob compute the shared secret key independently 
// Alice calculates s = B^a mod p (using Bob's public key and her private key)
 BigInteger secretKeyAlice = B.modPow(a, p);
 System.out.println("\nSecret Key computed by Alice: " + secretKeyAlice);
 // Bob calculates s = A^b mod p (using Alice's public key and his private key)
 BigInteger secretKeyBob = A.modPow(b, p);
 System.out.println("Secret Key computed by Bob: " + secretKeyBob);
 // Verification
 if (secretKeyAlice.equals(secretKeyBob)) {
 System.out.println("\nKey exchange successful! Shared secret is: " + secretKeyAlice);
 } else {
 System.out.println("\nError: Keys do not match.");
 }
 stdin.close();
 }
}
